```typescript
export function useEffectWithUnmountGuard<T>(
  effect: () => Promise<T> | (() => T),
  deps: React.DependencyList,
): void {
  const mountedRef = useRef(true);
  useEffect(() => {
    mountedRef.current = true;
    let result: T | undefined;
    const cleanup = () => { mountedRef.current = false; };
    Promise.resolve(effect()).then(v => {
      if (mountedRef.current) result = v;
    });
    return cleanup;
  }, deps);
  // 開発環境用: setState ヘルパー
  const safeSetState: typeof setState = useCallback((state) => {
    if (!mountedRef.current && process.env.NODE_ENV !== 'production') {
      console.warn('[unmount-guard] setState after unmount detected');
      return;
    }
    setState(state);
  }, []);
}
```
